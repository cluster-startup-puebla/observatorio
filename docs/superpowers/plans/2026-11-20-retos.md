### RLS `challenges`

1. **Empresario own:** ALL donde `company_id` = company del `auth.uid()` (`companies.profile_id = auth.uid()`). No puede poner `status = oculto` (solo admin). Puede `borrador` → `publicado` → `cerrado`.
2. **Anon / authenticated public portal:** SELECT donde `status = 'publicado' AND visibility = 'portal'`.
3. **Superadmin:** ALL via `private.is_superadmin()`.

`visibility = red` nunca sale en portal; solo en `/app/challenges` del dueño.

// src/lib/challenges/validate.ts
import { isChallengeArea } from "./areas";
import type { ChallengeUrgency, ChallengeVisibility } from "./status";

export type ChallengeInput = {
  name: string;
  area: string;
  descriptionHtml: string;
  successMetric: string;
  urgency: ChallengeUrgency | "" | null;
  visibility: ChallengeVisibility;
  showCompany: boolean;
};

export function validateChallengeInput(input: ChallengeInput): { ok: true } | { ok: false; error: string } {
  const name = input.name.trim();
  if (name.length < 8 || name.length > 80) {
    return { ok: false, error: "El nombre debe tener entre 8 y 80 caracteres." };
  }
  if (!isChallengeArea(input.area)) {
    return { ok: false, error: "Elige un área válida." };
  }
  const plain = input.descriptionHtml.replace(/<[^>]+>/g, "").replace(/\s+/g, "");
  if (plain.length < 8) {
    return { ok: false, error: "La descripción es demasiado corta." };
  }
  const meta = input.successMetric.trim();
  if (meta.length < 8 || meta.length > 500) {
    return { ok: false, error: "La meta debe tener entre 8 y 500 caracteres." };
  }
  if (input.visibility !== "portal" && input.visibility !== "red") {
    return { ok: false, error: "Visibilidad inválida." };
  }
  if (input.urgency && !["baja", "media", "alta"].includes(input.urgency)) {
    return { ok: false, error: "Urgencia inválida." };
  }
  return { ok: true };
}
```

-- supabase/migrations/20261120140000_challenges.sql

create table public.challenges (
  id uuid primary key default gen_random_uuid(),
  company_id uuid not null references public.companies (id) on delete cascade,
  created_by uuid not null references public.profiles (id),
  name text not null check (char_length(name) between 8 and 80),
  area text not null check (area in (
    'rrhh', 'finanzas', 'contabilidad', 'almacen', 'produccion',
    'compras', 'ventas', 'logistica', 'ti', 'calidad', 'otro'
  )),
  description_html text not null,
  success_metric text not null check (char_length(success_metric) between 8 and 500),
  sector_id uuid not null references public.sectors (id),
  urgency text check (urgency is null or urgency in ('baja', 'media', 'alta')),
  visibility text not null default 'red' check (visibility in ('portal', 'red')),
  show_company boolean not null default false,
  company_name_snapshot text check (
    company_name_snapshot is null
    or char_length(company_name_snapshot) between 2 and 160
  ),
  status text not null default 'borrador' check (
    status in ('borrador', 'publicado', 'oculto', 'cerrado')
  ),
  published_at timestamptz,
  hidden_at timestamptz,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index challenges_company_id_idx on public.challenges (company_id);
create index challenges_portal_idx
  on public.challenges (published_at desc)
  where status = 'publicado' and visibility = 'portal';

create trigger challenges_updated_at
  before update on public.challenges
  for each row execute function private.set_updated_at();

create table public.challenge_applications (
  id uuid primary key default gen_random_uuid(),
  challenge_id uuid not null references public.challenges (id) on delete cascade,
  applicant_type text not null check (
    applicant_type in ('startup', 'incubadora', 'emprendedor')
  ),
  applicant_profile_id uuid references public.profiles (id) on delete set null,
  pitch text not null default '',
  status text not null default 'recibida' check (
    status in ('recibida', 'en_revision', 'aceptada', 'rechazada')
  ),
  created_at timestamptz not null default now()
);

create index challenge_applications_challenge_id_idx
  on public.challenge_applications (challenge_id);

alter table public.challenges enable row level security;
alter table public.challenge_applications enable row level security;

create policy challenges_own on public.challenges
  for all to authenticated
  using (
    (select private.is_superadmin())
    or company_id in (
      select id from public.companies where profile_id = (select auth.uid())
    )
  )
  with check (
    (select private.is_superadmin())
    or (
      company_id in (
        select id from public.companies where profile_id = (select auth.uid())
      )
      and created_by = (select auth.uid())
      and status in ('borrador', 'publicado', 'cerrado')
    )
  );

create policy challenges_portal_select on public.challenges
  for select to anon, authenticated
  using (status = 'publicado' and visibility = 'portal');

create policy challenges_superadmin_all on public.challenges
  for all to authenticated
  using ((select private.is_superadmin()))
  with check ((select private.is_superadmin()));

create policy challenge_applications_superadmin_select
  on public.challenge_applications
  for select to authenticated
  using ((select private.is_superadmin()));

comment on table public.challenge_applications is
  'Stub v1: applications UI comes later. Startups/incubadoras/emprendedores.';
```

Use `private.set_updated_at()` — same function as `companies_updated_at` in `20261022000820_observatory_normalized_schema.sql`.

Note: empresario `with check` blocks writing `oculto`; only the superadmin branch allows that status. When the empresario SELECTs a row that is already `oculto`, RLS `using` still allows read; UPDATE fails unless superadmin — correct (company sees «oculto por moderación» read-only).

import { revalidatePath } from "next/cache";
import { validateChallengeInput } from "@/lib/challenges/validate";
import { sanitizeChallengeHtml } from "@/lib/html/sanitize-challenge-html";

export async function saveChallengeDraft(form: {

  });

  const company = await db
    .from("companies")
    .select("id, sector_id, legal_name")
    .eq("profile_id", account.id)

  const fields = {

    description_html: html,
    success_metric: form.successMetric.trim(),
    sector_id: company.data.sector_id,
    urgency: form.urgency || null,
    visibility: form.visibility,
    show_company: form.showCompany,
  };

  // Draft save never touches publicado/cerrado/oculto rows — only insert or update borrador.

        .update(fields)
        .eq("id", form.id)

        ...fields,
        status: "borrador",

// src/lib/admin/retos/list.ts
import "server-only";

import { adminDb } from "@/lib/admin/admin-db";
import { pageQuery } from "@/lib/admin/page-query";

export async function listRetos(rawPage: string | undefined) {
  const page = pageQuery(rawPage);
  const db = await adminDb();
  if (!db) return { page: page.page, rows: [] as Record<string, string>[], total: 0 };
  const counted = await db.from("challenges").select("id", { count: "exact", head: true });
  if (counted.error) throw new Error(counted.error.message);
  const result = await db
    .from("challenges")
    .select("id, name, area, status, visibility, show_company, company_name_snapshot, published_at, companies(legal_name)")
    .order("updated_at", { ascending: false })
    .range(page.from, page.to);
  if (result.error) throw new Error(result.error.message);
  const rows = (result.data ?? []).map((row) => {
    const company = row.companies as { legal_name?: string } | null;
    return {
      id: row.id,
      name: row.name,
      area: row.area,
      status: row.status,
      visibility: row.visibility,
      show_company: row.show_company ? "Sí" : "No",
      empresa: company?.legal_name ?? row.company_name_snapshot ?? "—",
      published_at: row.published_at ?? "",
    };
  });
  return { page: page.page, rows, total: counted.count ?? 0 };
}

// src/lib/admin/retos/set-status.ts
"use server";

import { revalidatePath } from "next/cache";
import { requireSuperadmin } from "@/lib/auth/require-superadmin";
import { createSupabaseWithAccess, readAccessToken } from "@/lib/supabase/auth-session";

export async function setChallengeModeration(
  id: string,
  next: "oculto" | "publicado",
): Promise<string | null> {
  await requireSuperadmin();
  const token = await readAccessToken();
  if (!token) return "No se pudo actualizar.";
  const db = createSupabaseWithAccess(token);
  const patch =
    next === "oculto"
      ? { status: "oculto" as const, hidden_at: new Date().toISOString() }
      : { status: "publicado" as const, hidden_at: null, published_at: new Date().toISOString() };
  const result = await db.from("challenges").update(patch).eq("id", id);
  if (result.error) return "No se pudo actualizar.";
  revalidatePath("/admin/retos");
  revalidatePath("/retos");
  return null;
}

# Lavas (loyalty / gamificación) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Sustituir el chip «0 lavas / Próximamente» por un ledger real de loyalty: el empresario ve saldo y tabla de valores; el servidor otorga lavas al completar perfil, terminar evaluación, marcar «Llegó», aceptar innovación abierta y publicar un reto; el superadmin edita catálogo, boost EIA y puede ajustar / barrer histórico.

**Architecture:** Tres tablas (`lava_actions`, `lava_boost_config`, `lava_ledger`) + RPC `award_lavas` (security definer) que calcula monto, aplica boost EIA (multiplicador + bono fijo) y escribe filas inmutables con origen `live` \| `backfill` \| `manual`. El cliente solo lee. Los hooks viven en las acciones/rutas que ya escriben el hecho (perfil, envío metodología, mesa, consent, y `awardLavasForChallengePublish` llamado desde el plan de retos). El saldo es `sum(amount)` del ledger por `company_id`.

**Tech Stack:** Next.js App Router, Supabase Postgres + RLS + RPC, Vitest, shadcn Sheet/Tooltip/toast (`sonner`). Cada archivo `.ts` / `.tsx` ≤ 90 líneas.

**Fuente de verdad de producto:** `docs/planeacion-lavas.md` (decisiones cerradas 20 nov 2026). Hermana: `docs/planeacion-retos.md`.

**Coordinación con retos:** Ambos planes usan el mismo export:

```ts
// src/lib/lavas/award-challenge-publish.ts
export async function awardLavasForChallengePublish(input: {
  companyId: string;
  challengeId: string;
  origen?: "live" | "backfill";
}): Promise<{ awarded: number } | { skipped: string }>
```

El plan de retos, al publicar (estado visible, no borrador), llama esta función. Ocultar un reto **no** llama a ninguna reversión.

## Decisiones cerradas (no renegociar)

1. Lavas = solo gamificación; **no** gate ni definen EIA.
2. «Llegó» (mesa QR → `arrivals` + CRM `llegar`) sí otorga.
3. Sin topes mensuales; solo flags «una vez» / `ref_key` único.
4. Boost EIA = `round(base * multiplier) + flat_bonus` cuando la empresa tiene consentimiento con `accepted_at` **y** la acción tiene `eia_boost = true`.
5. Tras instalar: barrido `origen = backfill` de hechos históricos.
6. Superadmin edita catálogo + boost en UI.
7. Tabla pública de valores en la app del empresario.
8. Chip con saldo real; tooltip sin narrativa «juntar para ser EIA».
9. Solo servidor otorga; ledger inmutable; `origen` obligatorio.
10. Ocultar reto no revierte lavas.
11. UI en español; toast/Alert, nunca `alert()`; archivos ≤ 90 líneas; Vitest.

```sql
create or replace function public.award_lavas(
  p_company_id uuid,
  p_action_code text,
  p_ref_key text,
  p_origen text,
  p_note text default null,
  p_manual_amount integer default null
)
returns integer
language plpgsql
security definer
set search_path = public
as $$
declare
  uid uuid := auth.uid();
  base integer;
  boost boolean;
  once_flag boolean;
  mult numeric;
  flat integer;
  is_eia boolean;
  awarded integer;
  inserted integer;
begin
  if uid is null then
    raise exception 'no autenticado';
  end if;
  if p_origen not in ('live', 'backfill', 'manual') then
    raise exception 'origen';
  end if;
  if p_origen = 'manual' and not (select private.is_superadmin()) then
    raise exception 'solo superadmin';
  end if;
  if p_origen = 'backfill' and not (select private.is_superadmin()) then
    raise exception 'solo superadmin';
  end if;

  select a.base_value, a.eia_boost, a.once
    into base, boost, once_flag
  from public.lava_actions a
  where a.code = p_action_code and a.active = true;

  if base is null and p_action_code <> 'ajuste_manual' then
    raise exception 'accion';
  end if;

  if p_action_code = 'ajuste_manual' then
    if p_manual_amount is null then
      raise exception 'monto';
    end if;
    awarded := p_manual_amount;
  else
    select c.multiplier, c.flat_bonus into mult, flat
    from public.lava_boost_config c where c.id = 1;

    select exists (
      select 1
      from public.companies co
      join public.innovation_consents ic on ic.profile_id = co.profile_id
      where co.id = p_company_id
        and ic.accepted_at is not null
    ) into is_eia;

    if boost and is_eia then
      awarded := round(base * coalesce(mult, 1))::integer + coalesce(flat, 0);
    else
      awarded := base;
    end if;
  end if;

  insert into public.lava_ledger
    (company_id, action_code, amount, ref_key, origen, note, created_by)
  values
    (p_company_id, p_action_code, awarded, p_ref_key, p_origen, p_note, uid)
  on conflict (company_id, action_code, ref_key) do nothing;

  get diagnostics inserted = row_count;
  if inserted = 0 then
    return 0;
  end if;
  return awarded;
end;
$$;

revoke all on function public.award_lavas(uuid, text, text, text, text, integer) from public, anon;
grant execute on function public.award_lavas(uuid, text, text, text, text, integer) to authenticated;
```

Notas fijas:
- Idempotencia: conflicto → `0` (ya otorgado).
- `ajuste_manual` usa `p_manual_amount` (puede ser negativo) y `ref_key` único (`crypto.randomUUID()` en el caller).
- En Task 3, `lava_ledger.amount` es `integer not null` **sin** `check (amount >= 0)` para permitir correcciones manuales.

```ts
// src/lib/lavas/compute-amount.ts
export function computeLavaAmount(input: {
  base: number;
  eiaBoost: boolean;
  isEia: boolean;
  multiplier: number;
  flatBonus: number;
}) {
  if (!input.eiaBoost || !input.isEia) return input.base;
  return Math.round(input.base * input.multiplier) + input.flatBonus;
}
```

```ts
// src/lib/lavas/award.ts
import "server-only";
import type { SupabaseClient } from "@supabase/supabase-js";
import type { LavaCode } from "./codes";

export type AwardOrigen = "live" | "backfill" | "manual";

export async function awardLavas(
  db: SupabaseClient,
  input: {
    companyId: string;
    actionCode: LavaCode;
    refKey: string;
    origen: AwardOrigen;
    note?: string;
    manualAmount?: number;
  },
) {
  const result = await db.rpc("award_lavas", {
    p_company_id: input.companyId,
    p_action_code: input.actionCode,
    p_ref_key: input.refKey,
    p_origen: input.origen,
    p_note: input.note ?? null,
    p_manual_amount: input.manualAmount ?? null,
  });
  if (result.error) {
    console.error("award_lavas", result.error.message);
    return { ok: false as const, reason: "rpc" };
  }
  const awarded = Number(result.data ?? 0);
  return { ok: true as const, awarded };
}
```

```sql
-- Inserta directo en lava_ledger (no llama award_lavas: ahí auth.uid() es null en el accept anónimo).
create or replace function public.award_lavas_for_consent(raw_token text)
returns integer
language plpgsql
security definer
set search_path = public
as $$
declare
  pid uuid;
  cid uuid;
  base integer;
  boost boolean;
  mult numeric;
  flat integer;
  awarded integer;
  inserted integer;
begin
  if raw_token is null
    or char_length(raw_token) < 43
    or char_length(raw_token) > 80
    or raw_token !~ '^[A-Za-z0-9_-]+$' then
    return 0;
  end if;

  select ic.profile_id into pid
  from public.innovation_consents ic
  where ic.token = raw_token
    and ic.accepted_at is not null;
  if pid is null then return 0; end if;

  select c.id into cid from public.companies c where c.profile_id = pid;
  if cid is null then return 0; end if;

  select a.base_value, a.eia_boost into base, boost
  from public.lava_actions a
  where a.code = 'innovacion_aceptada' and a.active = true;
  if base is null then return 0; end if;

  select c.multiplier, c.flat_bonus into mult, flat
  from public.lava_boost_config c where c.id = 1;

  -- Al aceptar, la empresa ya es EIA: aplica boost si la acción lo pide.
  if boost then
    awarded := round(base * coalesce(mult, 1))::integer + coalesce(flat, 0);
  else
    awarded := base;
  end if;

  insert into public.lava_ledger
    (company_id, action_code, amount, ref_key, origen, created_by)
  values
    (cid, 'innovacion_aceptada', awarded, pid::text, 'live', pid)
  on conflict (company_id, action_code, ref_key) do nothing;

  get diagnostics inserted = row_count;
  if inserted = 0 then return 0; end if;
  return awarded;
end;
$$;

revoke all on function public.award_lavas_for_consent(text) from public;
grant execute on function public.award_lavas_for_consent(text) to anon, authenticated;
```
